// Étape 1 : Quand on clique sur Encaisser, on ouvre directement le pavé PIN function lancerPaiementParPin() { if (panier.length === 0) { alert("Le panier est vide !"); return; } ouvrirPaveNumerique("", function(pinSaisi) { rechercherJoueurParPin(pinSaisi); }); } // Étape 2 : Recherche du ou des joueurs correspondant au PIN function rechercherJoueurParPin(pinSaisi) { const correspondances = comptes.filter(c => String(c.pin).padStart(4, '0') === String(pinSaisi).padStart(4, '0')); if (correspondances.length === 0) { afficherErreur("❌ Aucun compte ne correspond à ce code PIN."); } else if (correspondances.length === 1) { // Un seul joueur trouvé : demande de confirmation directe afficherConfirmationIdentite(correspondances[0]); } else { // Plusieurs joueurs ont le même PIN : on lui demande de toucher son nom afficherChoixMultiples(correspondances); } } // Étape 3 : Fenêtre "Est-ce bien vous ?" function afficherConfirmationIdentite(joueur) { const nouveauSolde = joueur.solde - totalPanier; // Vérification du seuil négatif autorisé avant même de valider if (nouveauSolde < joueur.seuil) { afficherAlerteSeuil(joueur, nouveauSolde); return; } const modale = document.getElementById("modale-confirmation-id"); document.getElementById("conf-nom-joueur").innerText = joueur.name; document.getElementById("conf-categorie").innerText = joueur.cat; document.getElementById("conf-solde-actuel").innerText = joueur.solde.toFixed(2) + " €"; document.getElementById("conf-montant-panier").innerText = "-" + totalPanier.toFixed(2) + " €"; document.getElementById("conf-nouveau-solde").innerText = nouveauSolde.toFixed(2) + " €"; // Bouton OUI document.getElementById("btn-oui-id").onclick = function() { modale.style.display = "none"; validerTransactionFinale(joueur); // Enregistre et synchronise avec le Sheet }; // Bouton NON document.getElementById("btn-non-id").onclick = function() { modale.style.display = "none"; }; modale.style.display = "flex"; }
